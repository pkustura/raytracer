cmake_minimum_required(VERSION 3.10)
project(Raytracer)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)  # lets editors/clangd find include paths and flags

# Default to an optimized build; the renderer runs on the CPU every frame.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Find packages
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 REQUIRED)

# Add GLEW package
find_package(GLEW REQUIRED)

set(GLEW_INCLUDE_DIRS "/opt/homebrew/opt/glew/include")
set(GLEW_LIBRARIES "/opt/homebrew/opt/glew/lib/libGLEW.dylib")

# Include directories
include_directories(src ${GLEW_INCLUDE_DIRS})

# Rendering core shared by the interactive app and the benchmark
add_library(RaytracerCore STATIC
    src/camera.cpp
    src/light.cpp
    src/ray.cpp
    src/vector.cpp
    src/color.cpp
    src/material.cpp
    src/shape.cpp
    src/obj_loader.cpp
    src/mesh.cpp
    src/renderer.cpp
    src/scene.cpp
    src/benchmark.cpp
    src/pathtracer.cpp
)

# Absolute path to the models, so the programs can be run from any directory
target_compile_definitions(RaytracerCore PRIVATE RAYTRACER_ASSET_DIR="${CMAKE_SOURCE_DIR}/objs")

# Download the models (objs/) during the build if any are missing.
# Turn off with -DRAYTRACER_FETCH_MODELS=OFF to build offline.
option(RAYTRACER_FETCH_MODELS "Download missing models with fetch_models.sh during the build" ON)
if(RAYTRACER_FETCH_MODELS)
    # Runs on every build. The script only downloads models that are missing,
    # so when everything is present it just checks and prints one line.
    add_custom_target(fetch_models ALL
        COMMAND ${CMAKE_SOURCE_DIR}/fetch_models.sh
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
        VERBATIM
    )
endif()

# Dear ImGui for the in-app UI (fetched at configure time, pinned to a release)
include(FetchContent)
FetchContent_Declare(imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG v1.92.9b
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(imgui)

add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl2.cpp
)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC glfw OpenGL::GL)

# Interactive app
add_executable(Raytracer src/main.cpp src/app_ui.cpp)

# Define GL_SILENCE_DEPRECATION to silence OpenGL deprecation warnings on macOS
target_compile_definitions(Raytracer PRIVATE GL_SILENCE_DEPRECATION)

# Link libraries
target_link_libraries(Raytracer RaytracerCore imgui OpenGL::GL glfw ${GLEW_LIBRARIES})

# Headless benchmark
add_executable(Benchmark bench/benchmark.cpp)
target_link_libraries(Benchmark RaytracerCore)

# Make sure the models are there before the programs that load them are built
if(RAYTRACER_FETCH_MODELS)
    add_dependencies(Raytracer fetch_models)
    add_dependencies(Benchmark fetch_models)
endif()
